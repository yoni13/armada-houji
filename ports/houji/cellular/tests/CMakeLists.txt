cmake_minimum_required(VERSION 3.16)
project(houji_cellular_ui_tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Xml)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MM REQUIRED IMPORTED_TARGET ModemManager)
if(NOT MMQT_SOURCE)
    message(FATAL_ERROR "Set MMQT_SOURCE to the pinned, patched modemmanager-qt tree")
endif()
file(GLOB sources "${MMQT_SOURCE}/src/*.cpp" "${MMQT_SOURCE}/src/dbus/*.cpp")
file(GLOB headers "${MMQT_SOURCE}/src/*.h" "${MMQT_SOURCE}/src/dbus/*.h")
# Tests use the session-bus fake service name; no host modem is accessed.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/modemmanagerqt_export.h" "#pragma once\n#define MODEMMANAGERQT_EXPORT\n#define MODEMMANAGERQT_NO_EXPORT\n")
add_executable(lifecycle lifecycle.cpp ${sources} ${headers})
target_compile_definitions(lifecycle PRIVATE MMQT_STATIC QT_NO_KEYWORDS)
target_include_directories(lifecycle PRIVATE "${MMQT_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(lifecycle PRIVATE Qt6::Core Qt6::DBus Qt6::Xml PkgConfig::MM)
